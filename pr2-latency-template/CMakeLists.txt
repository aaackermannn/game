cmake_minimum_required(VERSION 3.16)
project(pr2_latency_template CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_CXX_EXTENSIONS OFF)
add_library(protocol src/protocol.cpp)
add_library(telemetry src/telemetry.cpp)
add_library(reliability src/reliability/reliable_channel.cpp src/reliability/adaptive_timeout.cpp)
add_library(transport src/transport/udp_socket.cpp)
foreach(module protocol telemetry reliability transport)
    target_include_directories(${module} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
    if(MSVC)
        target_compile_options(${module} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${module} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(WIN32)
    target_link_libraries(transport PUBLIC ws2_32)
endif()
add_executable(server src/server_main.cpp)
target_link_libraries(server PRIVATE protocol reliability transport)
add_executable(client src/client_main.cpp src/reliability_client.cpp)
target_link_libraries(client PRIVATE protocol telemetry reliability transport)
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
